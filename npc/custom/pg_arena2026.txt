//===== rAthena Script =======================================
//= Arena 7 vs 7 (custom, playground)
//===== Description: =========================================
//= Two pre-made parties of 7 fight best-of-3 rounds on the 2026arena1..4 maps (one match per map, up to 4 at once).
//=  * Each party must have exactly 7 online members standing in the lobby map, with no repeated job line inside the party
//=    (job line = the class without its upper / baby / third / fourth flag: Cardinal and Arch Bishop are the same line).
//=  * The leader registers at the Arena Master; when two parties are waiting and an arena is free the match starts.
//=  * A round is won by the party that eliminates the other (dead, offline or out of the map counts as eliminated).
//=    When the round time runs out the party with more survivors wins, then the one with more HP left; still equal = draw.
//=  * Between rounds everybody is revived, healed, cleaned of status effects and warped back to its start; sides swap each round.
//=  * The first party to win 2 rounds wins the match (a draw is replayed; at most 5 rounds).
//=  * At registration a party can pick a specific arena or "any free arena"; the Arena Master's own name shows the queue size,
//=    and its "Arena status" option lists every arena (phase, score, round, survivors) and every queued party.
//=  * A live scoreboard is announced inside the arena map whenever the number of survivors on either side changes.
//= Everything you may want to tune is at the top of OnInit of Arena2026Ctrl.
//============================================================

// -----------------------------------------------------------
// Returns "" when the party can play, otherwise the reason it can not.
//   getarg(0) party id   getarg(1) members required   getarg(2) lobby map
// -----------------------------------------------------------
// Label for a party, guild-first: the leader's guild name when they have one (online), else the party name.
function	script	F_GuildTag	{
	.@pid = getarg(0);
	.@lid = getpartyleader(.@pid, 1);
	if (.@lid > 0 && isloggedin(.@lid)) {
		.@me = playerattached();
		if (attachrid(.@lid)) {
			.@gid = getcharid(2);
			if (.@gid)
				.@n$ = getguildname(.@gid);
			detachrid;
		}
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return (.@n$ != "") ? .@n$ : getpartyname(.@pid);
}

// Short label for a party: "Name" or "Name [GuildTag]" when the leader is online and in a guild.
function	script	F_PartyLabel	{
	.@pid = getarg(0);
	.@n$ = getpartyname(.@pid);
	.@lid = getpartyleader(.@pid, 1);
	if (.@lid > 0 && isloggedin(.@lid)) {
		.@me = playerattached();
		if (attachrid(.@lid)) {
			.@gid = getcharid(2);
			if (.@gid)
				.@n$ += " [" + getguildname(.@gid) + "]";
			detachrid;
		}
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return .@n$;
}

function	script	F_ArenaCheck	{
	.@pid = getarg(0);
	.@need = getarg(1);
	.@lobby$ = getarg(2);
	if (!.@pid)
		return "The party does not exist any more.";
	.@me = playerattached();
	getpartymember .@pid, 2;
	.@n = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	if (.@n != .@need)
		return "The party needs exactly " + .@need + " members (it has " + .@n + ").";
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!isloggedin(.@aid[.@i]))
			return "All the members must be online.";
	}
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aid[.@i])) {
			.@err$ = "All the members must be online.";
			break;
		}
		if (strcharinfo(3) != .@lobby$) {
			.@err$ = strcharinfo(0) + " is not in the lobby.";
			break;
		}
		if (isdead()) {
			.@err$ = strcharinfo(0) + " is dead.";
			break;
		}
		.@line[.@i] = eaclass() & EAJ_UPPERMASK;
		.@jname$[.@i] = jobname(Class);
		.@pname$[.@i] = strcharinfo(0);
	}
	if (.@me)
		attachrid(.@me);
	else
		detachrid;
	if (.@err$ != "")
		return .@err$;
	for (.@i = 0; .@i < .@n; .@i++) {
		for (.@j = .@i + 1; .@j < .@n; .@j++) {
			if (.@line[.@i] == .@line[.@j])
				return "Repeated job line: " + .@pname$[.@i] + " (" + .@jname$[.@i] + ") and " + .@pname$[.@j] + " (" + .@jname$[.@j] + ").";
		}
	}
	return "";
}

// The lobby NPC: register / cancel / status

hero_ent3,73,128,4	script	Arena Master	858,{
	mes "[Arena Master]";
	mes "7 vs 7, best of 3 rounds.";
	mes "One party of 7 members, every member of a different job line.";
	next;
	switch (select("Register my party:Cancel my registration:Arena status:Rules:Leave")) {
	case 1:
		.@pid = getcharid(1);
		mes "[Arena Master]";
		if (!.@pid) {
			mes "You need a party.";
			close;
		}
		if (!is_party_leader()) {
			mes "Only the party leader can register the party.";
			close;
		}
		if (callsub(L_Busy, .@pid)) {
			mes "Your party is already registered or fighting.";
			close;
		}
		.@e$ = callfunc("F_ArenaCheck", .@pid, 7, "hero_ent3");
		if (.@e$ != "") {
			mes "Your party can not register:";
			mes "^FF0000" + .@e$ + "^000000";
			close;
		}
		mes "Your party is valid. Pick an arena, or let the system choose:";
		next;
		.@any$ = "";		// guild tags of the parties queued with "any free arena"
		for (.@i = 0; .@i < $@arq_n; .@i++) {
			if ($@arq_pref[.@i] == 0)
				.@any$ += ((.@any$ != "") ? ", " : "") + callfunc("F_GuildTag", $@arq_pid[.@i]);
		}
		.@m$ = "Any free arena (" + $@arq_n + " waiting" + ((.@any$ != "") ? ": " + .@any$ : "") + ")";
		for (.@s = 1; .@s <= 4; .@s++) {
			.@m$ += ":Arena " + .@s + " - ";
			if ($@ar_state[.@s]) {
				.@m$ += "busy, " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " vs " + callfunc("F_GuildTag", $@ar_pb[.@s]);
			} else {
				.@w$ = "";		// guild tags of the parties queued specifically for this arena
				for (.@i = 0; .@i < $@arq_n; .@i++) {
					if ($@arq_pref[.@i] == .@s)
						.@w$ += ((.@w$ != "") ? ", " : "") + callfunc("F_GuildTag", $@arq_pid[.@i]);
				}
				.@m$ += "free" + ((.@w$ != "") ? " (" + .@w$ + " waiting)" : "");
			}
		}
		.@sel = select(.@m$ + ":Cancel");
		mes "[Arena Master]";
		if (.@sel == 6) {
			mes "Registration cancelled.";
			close;
		}
		$@arq_pid[$@arq_n] = .@pid;
		$@arq_pref[$@arq_n] = (.@sel == 1) ? 0 : .@sel - 1;		// 0 = any arena, else the arena number
		$@arq_n++;
		setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
		mes "Your party is registered (position " + $@arq_n + "), " + (($@arq_pref[$@arq_n - 1] == 0) ? "any free arena" : ("arena " + $@arq_pref[$@arq_n - 1] + " only")) + ".";
		mes "Stay in the lobby, all together. When another party is ready and an arena is free, you will be warped.";
		announce "[Arena] " + getpartyname(.@pid) + " is registered and waiting for an opponent (queue: " + $@arq_n + ").", bc_all, 0x00FFAA;
		close2;
		donpcevent "Arena2026Ctrl::OnQueueCheck";
		end;
	case 2:
		.@pid = getcharid(1);
		mes "[Arena Master]";
		if (!.@pid || !is_party_leader()) {
			mes "Only the party leader can cancel.";
			close;
		}
		.@i = inarray($@arq_pid, .@pid);
		if (.@i < 0 || .@i >= $@arq_n) {
			mes "Your party is not waiting in the queue.";
			close;
		}
		deletearray $@arq_pid[.@i], 1;
		deletearray $@arq_pref[.@i], 1;
		$@arq_n--;
		setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
		mes "Registration cancelled.";
		close;
	case 3:
		mes "[Arena Master]";
		mes "^0000FFArenas:^000000";
		for (.@s = 1; .@s <= 4; .@s++) {
			if (!$@ar_state[.@s]) {
				mes "Arena " + .@s + ": free";
			} else {
				.@phase$ = ($@ar_state[.@s] == 1) ? "preparing" : (($@ar_state[.@s] == 2) ? "fighting" : "between rounds");
				mes "Arena " + .@s + " [" + callfunc("F_GuildTag", $@ar_pa[.@s]) + " vs " + callfunc("F_GuildTag", $@ar_pb[.@s]) + "] (" + .@phase$ + "): " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + ", round " + $@ar_round[.@s] + ", alive " + $@ar_alive_a[.@s] + "-" + $@ar_alive_b[.@s];
			}
		}
		mes "^0000FFQueue (" + $@arq_n + " waiting):^000000";
		if (!$@arq_n)
			mes "(empty)";
		for (.@i = 0; .@i < $@arq_n; .@i++) {
			.@w$ = ($@arq_pref[.@i] == 0) ? "any free arena" : ("Arena " + $@arq_pref[.@i] + " only");
			mes (.@i + 1) + ". [" + callfunc("F_GuildTag", $@arq_pid[.@i]) + " waiting] - " + .@w$;
		}
		close;
	case 4:
		mes "[Arena Master]";
		mes "- Best of 3 rounds: win 2 to win the match.";
		mes "- A round ends when a whole party is dead, offline or out of the arena.";
		mes "- If the time runs out: more survivors wins, then more HP left.";
		mes "- Everybody is revived and healed between rounds; sides swap.";
		mes "- No repeated job line inside a party (Cardinal = Arch Bishop).";
		close;
	case 5:
		close;
	}
	end;

L_Busy:
	.@p = getarg(0);
	if (inarray($@arq_pid, .@p) >= 0 && inarray($@arq_pid, .@p) < $@arq_n)
		return 1;
	for (.@s = 1; .@s <= 4; .@s++) {
		if ($@ar_state[.@s] && ($@ar_pa[.@s] == .@p || $@ar_pb[.@s] == .@p))
			return 1;
	}
	return 0;
}

// The controller: queue, matches, rounds. One tick per second.
// -----------------------------------------------------------
-	script	Arena2026Ctrl	FAKE_NPC,{
OnInit:
	// ---- settings ----
	.slots = 4;				// arenas (maps)
	.team = 7;				// players per party
	.wins = 2;				// rounds to win a match
	.maxrounds = 5;			// a match never lasts more than this many rounds
	.countdown = 25;		// seconds of preparation before every round (players can move inside their area and buff)
	.roundtime = 300;		// seconds per round
	.between = 8;			// seconds after a round before the next set-up
	.endwait = 10;			// seconds after the match before everybody goes back to the lobby
	.tp_allowed = 0;		// 1 = Teleport / Fly Wing allowed inside the arenas
	.lobby$ = "hero_ent3";
	.lx = 69;
	.ly = 136;
	setarray .map$[1], "2026arena1", "2026arena2", "2026arena3", "2026arena4";
	// start cell: every one of the 7 players of a side stacks on the very same, verified walkable cell (west / east)
	setarray .wx[1], 50, 34, 41, 203;
	setarray .wy[1], 96, 94, 95, 206;
	setarray .ex[1], 145, 165, 160, 296;
	setarray .ey[1], 96, 94, 95, 206;

	for (.@i = 1; .@i <= .slots; .@i++) {
		.@m$ = .map$[.@i];
		setmapflag .@m$, MF_GVG;
		setmapflag .@m$, MF_PVP_NOGUILD;		// (also honoured on GvG maps) guild mates / allies of the two parties still fight each other
		setmapflag .@m$, MF_NOMEMO;
		setmapflag .@m$, MF_NORETURN;
		setmapflag .@m$, MF_NOBRANCH;
		setmapflag .@m$, MF_NOPENALTY;
		setmapflag .@m$, MF_NOWARPTO;
		setmapflag .@m$, MF_NOTRADE;
		if (!.tp_allowed)
			setmapflag .@m$, MF_NOTELEPORT;
		setmapflagnosave .@m$, .lobby$, .lx, .ly;		// whoever logs out inside comes back in the lobby
		$@ar_state[.@i] = 0;
		callsub L_Walls, .@i, 1;
	}
	$@arq_n = 0;
	setnpcdisplay "Arena Master", "Arena Master (0 waiting)";
	initnpctimer;
	end;

OnTimer1000:
	if ($@arq_n >= 2)
		callsub L_Queue;
	for (.@s = 1; .@s <= .slots; .@s++) {
		if ($@ar_state[.@s])
			callsub L_Slot, .@s;
	}
	initnpctimer;
	end;

OnQueueCheck:
	callsub L_Queue;
	end;

// ---------------------------------------------------------
// queue -> match
// ---------------------------------------------------------
L_Queue:
	// matches every free arena against the first two queued parties compatible with it (preference 0 = any arena, or that arena number).
	// Restarts the scan whenever the queue changes, since a removal or a match shifts every index after it.
	.@changed = 1;
	while (.@changed) {
		.@changed = 0;
		for (.@s = 1; .@s <= .slots && !.@changed; .@s++) {
			if ($@ar_state[.@s])
				continue;
			.@i1 = -1;
			.@i2 = -1;
			for (.@i = 0; .@i < $@arq_n; .@i++) {
				if ($@arq_pref[.@i] != 0 && $@arq_pref[.@i] != .@s)
					continue;
				if (.@i1 < 0)
					.@i1 = .@i;
				else {
					.@i2 = .@i;
					break;
				}
			}
			if (.@i1 < 0 || .@i2 < 0)
				continue;
			.@p1 = $@arq_pid[.@i1];
			.@p2 = $@arq_pid[.@i2];
			.@e1$ = callfunc("F_ArenaCheck", .@p1, .team, .lobby$);
			.@e2$ = callfunc("F_ArenaCheck", .@p2, .team, .lobby$);
			if (.@e1$ != "" || .@e2$ != "") {
				if (.@e2$ != "") {		// remove the higher index first so .@i1 stays valid
					callsub L_Tell, .@p2, "Arena: your registration was cancelled - " + .@e2$;
					deletearray $@arq_pid[.@i2], 1;
					deletearray $@arq_pref[.@i2], 1;
					$@arq_n--;
				}
				if (.@e1$ != "") {
					callsub L_Tell, .@p1, "Arena: your registration was cancelled - " + .@e1$;
					deletearray $@arq_pid[.@i1], 1;
					deletearray $@arq_pref[.@i1], 1;
					$@arq_n--;
				}
				setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
				.@changed = 1;
				continue;
			}
			deletearray $@arq_pid[.@i2], 1;
			deletearray $@arq_pref[.@i2], 1;
			deletearray $@arq_pid[.@i1], 1;
			deletearray $@arq_pref[.@i1], 1;
			$@arq_n -= 2;
			setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
			callsub L_Begin, .@s, .@p1, .@p2;
			.@changed = 1;
		}
	}
	return;

L_Tell:
	.@a = getpartyleader(getarg(0), 1);
	if (.@a > 0 && isloggedin(.@a)) {
		.@me = playerattached();
		if (attachrid(.@a))
			dispbottom getarg(1);
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return;

	$@ar_pa[.@s] = getarg(1);
	$@ar_pb[.@s] = getarg(2);
	getpartymember getarg(1), 2;
	for (.@k = 0; .@k < .team; .@k++)
		$@ar_ma[.@s * 10 + .@k] = $@partymemberaid[.@k];
	getpartymember getarg(2), 2;
	for (.@k = 0; .@k < .team; .@k++)
		$@ar_mb[.@s * 10 + .@k] = $@partymemberaid[.@k];
	$@ar_round[.@s] = 1;
	$@ar_sc_a[.@s] = 0;
	$@ar_sc_b[.@s] = 0;
	$@ar_fin[.@s] = 0;
	announce "[Arena " + .@s + "] " + getpartyname(getarg(1)) + " vs " + getpartyname(getarg(2)) + " - best of 3!", bc_all, 0x00FFAA;
	callsub L_Prep, .@s;
	return;

// ---------------------------------------------------------
// set-up of a round: revive, heal, clean, warp to the start (sides swap every round), freeze, countdown
// ---------------------------------------------------------
L_Prep:
	.@s = getarg(0);
	callsub L_Walls, .@s, 0;
	.@flip = ($@ar_round[.@s] % 2 == 0) ? 1 : 0;
	.@on0 = 0;
	.@on1 = 0;
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (!.@a || !isloggedin(.@a))
				continue;
			if (!attachrid(.@a))
				continue;
			if (.@t == 0) .@on0++; else .@on1++;
			recovery 0, getcharid(0);
			sc_end SC_ALL;
			percentheal 100, 100;
			if ((.@t ^ .@flip) == 0)
				warp .map$[.@s], .wx[.@s], .wy[.@s];
			else
				warp .map$[.@s], .ex[.@s], .ey[.@s];
			// AP after the warp (a map change is the last thing that could touch it); healap goes through status_heal like every other heal
			healap MaxAp;
			detachrid;
		}
	}
	// a party that is gone loses by forfeit; nobody left ends the match
	if (!.@on0 || !.@on1) {
		$@ar_fin[.@s] = 1;
		if (!.@on0 && .@on1)
			$@ar_sc_b[.@s] = .wins;
		else if (.@on0 && !.@on1)
			$@ar_sc_a[.@s] = .wins;
		callsub L_Walls, .@s, 1;
		callsub L_EndMatch, .@s;
		return;
	}
	$@ar_alive_a[.@s] = .@on0;
	$@ar_alive_b[.@s] = .@on1;
	$@ar_state[.@s] = 1;
	$@ar_t[.@s] = .countdown;
	mapannounce .map$[.@s], "Preparation: " + .countdown + " seconds. Move only inside your area and use your buffs.", 0, 0x00FFFF;
	mapannounce .map$[.@s], "Round " + $@ar_round[.@s] + " - " + getpartyname($@ar_pa[.@s]) + " " + $@ar_sc_a[.@s] + " : " + $@ar_sc_b[.@s] + " " + getpartyname($@ar_pb[.@s]), 0, 0xFFFF00;
	return;

// ---------------------------------------------------------
// one tick of a running arena
// ---------------------------------------------------------
L_Slot:
	.@s = getarg(0);
	.@st = $@ar_state[.@s];
	if (.@st == 1) {
		if ($@ar_t[.@s] > 0) {
			if ($@ar_t[.@s] <= 5 || $@ar_t[.@s] % 5 == 0)
				mapannounce .map$[.@s], "The round starts in " + $@ar_t[.@s] + "...", 0, 0xFFFFFF;
			$@ar_t[.@s]--;
		} else {
			callsub L_StartRound, .@s;
		}
	} else if (.@st == 2) {
		$@ar_t[.@s]--;
		.@aa = callsub(L_Alive, .@s, 0);
		.@ab = callsub(L_Alive, .@s, 1);
		if (.@aa != $@ar_alive_a[.@s] || .@ab != $@ar_alive_b[.@s]) {		// live scoreboard: only announces when someone dies / reconnects
			$@ar_alive_a[.@s] = .@aa;
			$@ar_alive_b[.@s] = .@ab;
			mapannounce .map$[.@s], getpartyname($@ar_pa[.@s]) + " " + .@aa + " - " + .@ab + " " + getpartyname($@ar_pb[.@s]) + " alive", 0, 0x00FFFF;
		}
		if (!.@aa || !.@ab || $@ar_t[.@s] <= 0)
			callsub L_EndRound, .@s, .@aa, .@ab;
	} else if (.@st == 3) {
		if ($@ar_t[.@s] > 0) {
			$@ar_t[.@s]--;
		} else if ($@ar_fin[.@s]) {
			callsub L_Finish, .@s;
		} else {
			callsub L_Prep, .@s;
		}
	}
	return;

L_StartRound:
	.@s = getarg(0);
	callsub L_Walls, .@s, 1;
	$@ar_state[.@s] = 2;
	$@ar_t[.@s] = .roundtime;
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (.@a && isloggedin(.@a) && attachrid(.@a)) {
				showdigit .roundtime, 2;		// live MM:SS clock at the top of the screen, like a PVP room; the client ticks it down on its own
				detachrid;
			}
		}
	}
	mapannounce .map$[.@s], "FIGHT!", 0, 0xFF0000;
	return;

// members of a team that are alive and still inside the arena map
L_Alive:
	.@s = getarg(0);
	.@c = 0;
	for (.@k = 0; .@k < .team; .@k++) {
		.@a = callsub(L_Member, .@s, getarg(1), .@k);
		if (!.@a || !isloggedin(.@a) || isdead(.@a))
			continue;
		if (attachrid(.@a)) {
			if (strcharinfo(3) == .map$[.@s])
				.@c++;
			detachrid;
		}
	}
	return .@c;

// sum of HP% of the living members of a team (tie-break when the time runs out)
L_HpSum:
	.@s = getarg(0);
	.@sum = 0;
	for (.@k = 0; .@k < .team; .@k++) {
		.@a = callsub(L_Member, .@s, getarg(1), .@k);
		if (!.@a || !isloggedin(.@a) || isdead(.@a))
			continue;
		if (attachrid(.@a)) {
			if (strcharinfo(3) == .map$[.@s] && MaxHp > 0)
				.@sum += Hp * 100 / MaxHp;
			detachrid;
		}
	}
	return .@sum;

L_EndRound:
	.@s = getarg(0);
	.@aa = getarg(1);
	.@ab = getarg(2);
	for (.@t = 0; .@t <= 1; .@t++) {		// stop the round clock right as the round ends (it would otherwise keep counting past 0)
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (.@a && isloggedin(.@a) && attachrid(.@a)) {
				showdigit 0;
				detachrid;
			}
		}
	}
	.@win = 0;
	if (.@aa > .@ab)
		.@win = 1;
	else if (.@ab > .@aa)
		.@win = 2;
	else {
		.@ha = callsub(L_HpSum, .@s, 0);
		.@hb = callsub(L_HpSum, .@s, 1);
		if (.@ha > .@hb) .@win = 1;
		else if (.@hb > .@ha) .@win = 2;
	}
	if (.@win == 1) {
		$@ar_sc_a[.@s]++;
		mapannounce .map$[.@s], "Round " + $@ar_round[.@s] + ": " + getpartyname($@ar_pa[.@s]) + " wins the round!", 0, 0x00FF00;
	} else if (.@win == 2) {
		$@ar_sc_b[.@s]++;
		mapannounce .map$[.@s], "Round " + $@ar_round[.@s] + ": " + getpartyname($@ar_pb[.@s]) + " wins the round!", 0, 0x00FF00;
	} else {
		mapannounce .map$[.@s], "Round " + $@ar_round[.@s] + ": draw - the round is replayed.", 0, 0x00FF00;
	}
	// everybody stays where they are, but nobody can be hurt any more
	if ($@ar_sc_a[.@s] >= .wins || $@ar_sc_b[.@s] >= .wins || $@ar_round[.@s] >= .maxrounds) {
		callsub L_EndMatch, .@s;
	} else {
		$@ar_round[.@s]++;
		$@ar_state[.@s] = 3;
		$@ar_t[.@s] = .between;
		$@ar_fin[.@s] = 0;
	}
	return;

L_EndMatch:
	.@s = getarg(0);
	callsub L_Walls, .@s, 1;
	if ($@ar_sc_a[.@s] > $@ar_sc_b[.@s])
		.@msg$ = getpartyname($@ar_pa[.@s]) + " wins the match " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + "!";
	else if ($@ar_sc_b[.@s] > $@ar_sc_a[.@s])
		.@msg$ = getpartyname($@ar_pb[.@s]) + " wins the match " + $@ar_sc_b[.@s] + " - " + $@ar_sc_a[.@s] + "!";
	else
		.@msg$ = "The match ends in a draw " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + ".";
	announce "[Arena " + .@s + "] " + .@msg$, bc_all, 0x00FFAA;
	mapannounce .map$[.@s], .@msg$, 0, 0x00FF00;
	$@ar_fin[.@s] = 1;
	$@ar_state[.@s] = 3;
	$@ar_t[.@s] = .endwait;
	return;

// after the match: everybody back to the lobby, healed, and the arena is free again
L_Finish:
	.@s = getarg(0);
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (!.@a || !isloggedin(.@a) || !attachrid(.@a))
				continue;
			pcblockmove getcharid(3), 0;
			if (strcharinfo(3) == .map$[.@s]) {
				recovery 0, getcharid(0);
				sc_end SC_ALL;
				percentheal 100, 100;
				warp .lobby$, .lx, .ly;
				healap MaxAp;
			}
			detachrid;
		}
	}
	$@ar_state[.@s] = 0;
	$@ar_pa[.@s] = 0;
	$@ar_pb[.@s] = 0;
	callsub L_Queue;
	return;
}

//===== rAthena Script =======================================
//= Arena 7 vs 7 (custom, playground)
//===== Description: =========================================
//= Two pre-made parties of 7 fight best-of-3 rounds on the 2026arena1..4 maps (one match per map, up to 4 at once).
//=  * Each party must have exactly 7 online members standing in the lobby map, with no repeated job line inside the party
//=    (job line = the class without its upper / baby / third / fourth flag: Cardinal and Arch Bishop are the same line).
//=  * The leader registers at the Arena Master; when two parties are waiting and an arena is free the match starts.
//=  * A round is won by the party that eliminates the other (dead, offline or out of the map counts as eliminated).
//=    When the round time runs out the party with more survivors wins, then the one with more HP left; still equal = draw.
//=  * Between rounds everybody is revived, healed, cleaned of status effects and warped back to its start; sides swap each round.
//=  * The first party to win 2 rounds wins the match (a draw is replayed; at most 5 rounds).
//=  * At registration a party can pick a specific arena or "any free arena"; the Arena Master's own name shows the queue size,
//=    and its "Arena status" option lists every arena (phase, score, round, survivors) and every queued party.
//=  * A live scoreboard is announced inside the arena map whenever the number of survivors on either side changes.
//= Everything you may want to tune is at the top of OnInit of Arena2026Ctrl.
//============================================================

// -----------------------------------------------------------
// Returns "" when the party can play, otherwise the reason it can not.
//   getarg(0) party id   getarg(1) members required   getarg(2) lobby map
// -----------------------------------------------------------
// Label for a party, guild-first: the leader's guild name when they have one (online), else the party name.
function	script	F_GuildTag	{
	.@pid = getarg(0);
	.@lid = getpartyleader(.@pid, 1);
	if (.@lid > 0 && isloggedin(.@lid)) {
		.@me = playerattached();
		if (attachrid(.@lid)) {
			.@gid = getcharid(2);
			if (.@gid)
				.@n$ = getguildname(.@gid);
			detachrid;
		}
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return (.@n$ != "") ? .@n$ : getpartyname(.@pid);
}

// Short label for a party: "Name" or "Name [GuildTag]" when the leader is online and in a guild.
function	script	F_PartyLabel	{
	.@pid = getarg(0);
	.@n$ = getpartyname(.@pid);
	.@lid = getpartyleader(.@pid, 1);
	if (.@lid > 0 && isloggedin(.@lid)) {
		.@me = playerattached();
		if (attachrid(.@lid)) {
			.@gid = getcharid(2);
			if (.@gid)
				.@n$ += " [" + getguildname(.@gid) + "]";
			detachrid;
		}
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return .@n$;
}

function	script	F_ArenaCheck	{
	.@pid = getarg(0);
	.@need = getarg(1);
	.@lobby$ = getarg(2);
	if (!.@pid)
		return "The party does not exist any more.";
	.@me = playerattached();
	getpartymember .@pid, 2;
	.@n = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	if (.@n != .@need)
		return "The party needs exactly " + .@need + " members (it has " + .@n + ").";
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!isloggedin(.@aid[.@i]))
			return "All the members must be online.";
	}
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aid[.@i])) {
			.@err$ = "All the members must be online.";
			break;
		}
		if (strcharinfo(3) != .@lobby$) {
			.@err$ = strcharinfo(0) + " is not in the lobby.";
			break;
		}
		if (isdead()) {
			.@err$ = strcharinfo(0) + " is dead.";
			break;
		}
		.@line[.@i] = eaclass() & EAJ_UPPERMASK;
		.@jname$[.@i] = jobname(Class);
		.@pname$[.@i] = strcharinfo(0);
	}
	if (.@me)
		attachrid(.@me);
	else
		detachrid;
	if (.@err$ != "")
		return .@err$;
	for (.@i = 0; .@i < .@n; .@i++) {
		for (.@j = .@i + 1; .@j < .@n; .@j++) {
			if (.@line[.@i] == .@line[.@j])
				return "Repeated job line: " + .@pname$[.@i] + " (" + .@jname$[.@i] + ") and " + .@pname$[.@j] + " (" + .@jname$[.@j] + ").";
		}
	}
	return "";
}

// Sends a dispbottom message to a party's leader, if online, without disturbing whatever player (if any) is currently
// attached to the calling script. getarg(0) party id   getarg(1) message
function	script	F_Tell	{
	.@a = getpartyleader(getarg(0), 1);
	if (.@a > 0 && isloggedin(.@a)) {
		.@me = playerattached();
		if (attachrid(.@a))
			dispbottom getarg(1);
		if (.@me)
			attachrid(.@me);
		else
			detachrid;
	}
	return 0;
}

// Leaderboard: increments one column ("wins"/"losses"/"draws") of `arena_stats` for the given guild/party name,
// creating the row on first use. getarg(0) name   getarg(1) column
function	script	F_StatBump	{
	.@n$ = escape_sql(getarg(0));
	.@col$ = getarg(1);		// always one of our own literals below, never player input - safe to interpolate
	query_sql("INSERT INTO `arena_stats` (`name`,`" + .@col$ + "`) VALUES ('" + .@n$ + "',1) ON DUPLICATE KEY UPDATE `" + .@col$ + "` = `" + .@col$ + "` + 1");
	return 0;
}

// Spectator mode: clears the invisibility/action-block state, warps back to the lobby and drops the tracking entry.
// A no-op if the account is not currently spectating (safe to call unconditionally, e.g. on logout or match end).
// getarg(0) account id
function	script	F_SpecEnd	{
	.@aid = getarg(0);
	.@i = inarray($@spec_aid, .@aid);
	if (.@i < 0)
		return 0;
	deletearray $@spec_aid[.@i], 1;
	deletearray $@spec_s[.@i], 1;
	$@spec_n--;
	.@me = playerattached();
	if (attachrid(.@aid)) {
		setpcblock PCBLOCK_ALL, 0;
		setoption OPTION_INVISIBLE, 0;
		#ar_wasspec = 0;
		warp $@lobby$, $@lx, $@ly;
	}
	if (.@me)
		attachrid(.@me);
	else
		detachrid;
	return 0;
}

hero_ent3,73,128,4	script	Arena Master	858,{
	mes "[Arena Master]";
	mes "7 vs 7, best of 3 rounds.";
	mes "One party of 7 members, every member of a different job line.";
	next;
	// menu is built dynamically (each row's ACTION is tracked in parallel, not a fixed case number) since which rows
	// show up depends on the viewer (GM Tools only for GMs, Watch/Stop watching depending on current spectator state).
	// (leaderboard results still accumulate in `arena_stats` every match - just no in-game menu for it, a web page
	// will read that table later)
	.@myaid = getcharid(3);
	.@watching = inarray($@spec_aid, .@myaid);

	.@n = 0;
	.@opt$ = "Register my party";	.@act$[.@n] = "register";	.@n++;
	.@opt$ += ":Cancel my registration";	.@act$[.@n] = "cancel";	.@n++;
	.@opt$ += ":Arena status";	.@act$[.@n] = "status";	.@n++;
	.@opt$ += ":Rules";	.@act$[.@n] = "rules";	.@n++;
	if (.@watching >= 0) {
		.@opt$ += ":Stop watching";	.@act$[.@n] = "stopwatch";	.@n++;
	} else {
		.@opt$ += ":Watch a match";	.@act$[.@n] = "watch";	.@n++;
	}
	if (.@isgm) {
		.@opt$ += ":GM Tools";	.@act$[.@n] = "gmtools";	.@n++;
	}
	.@opt$ += ":Leave";	.@act$[.@n] = "leave";	.@n++;
	.@act$ = .@act$[select(.@opt$) - 1];
	if (.@act$ == "register") {
		.@pid = getcharid(1);
		mes "[Arena Master]";
		if (!.@pid) {
			mes "You need a party.";
			close;
		}
		if (!is_party_leader()) {
			mes "Only the party leader can register the party.";
			close;
		}
		if (callsub(L_Busy, .@pid)) {
			mes "Your party is already registered or fighting.";
			close;
		}
		.@e$ = callfunc("F_ArenaCheck", .@pid, 7, "hero_ent3");
		if (.@e$ != "") {
			mes "Your party can not register:";
			mes "^FF0000" + .@e$ + "^000000";
			close;
		}
		mes "Your party is valid. Pick an arena, or let the system choose:";
		next;
		.@m$ = "Any free arena [ ^FF1493Matchmaking^000000 ]";
		for (.@s = 1; .@s <= 4; .@s++) {
			.@m$ += ":^F77C85~^000000^0152B5 Arena " + .@s + "^000000 [ ";
			if ($@ar_state[.@s]) {
				.@m$ += "^64CE64" + callfunc("F_GuildTag", $@ar_pa[.@s]) + "^000000 vs ^64CE64" + callfunc("F_GuildTag", $@ar_pb[.@s]) + "^000000";
			} else {
				.@w$ = "";		// guild tags of the parties queued specifically for this arena
				for (.@i = 0; .@i < $@arq_n; .@i++) {
					if ($@arq_pref[.@i] == .@s)
						.@w$ += ((.@w$ != "") ? ", " : "") + callfunc("F_GuildTag", $@arq_pid[.@i]);
				}
				.@m$ += (.@w$ != "") ? ("^FF3A32" + .@w$ + "^000000 ^808080is waiting^000000") : "^808080free^000000";
			}
			.@m$ += " ]";
		}
		.@sel = select(.@m$ + ":^808080Close^000000");
		mes "[Arena Master]";
		if (.@sel == 6) {
			mes "Registration cancelled.";
			close;
		}
		$@arq_pid[$@arq_n] = .@pid;
		$@arq_pref[$@arq_n] = (.@sel == 1) ? 0 : .@sel - 1;		// 0 = any arena, else the arena number
		$@arq_n++;
		setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
		mes "Your party is registered (position " + $@arq_n + "), " + (($@arq_pref[$@arq_n - 1] == 0) ? "any free arena" : ("arena " + $@arq_pref[$@arq_n - 1] + " only")) + ".";
		mes "Stay in the lobby, all together. When another party is ready and an arena is free, you will be warped.";
		announce "[Arena] " + callfunc("F_GuildTag", .@pid) + " is registered and waiting for an opponent (queue: " + $@arq_n + ").", bc_all, 0x00FFAA;
		close2;
		donpcevent "Arena2026Ctrl::OnQueueCheck";
		end;
	} else if (.@act$ == "cancel") {
		.@pid = getcharid(1);
		mes "[Arena Master]";
		if (!.@pid || !is_party_leader()) {
			mes "Only the party leader can cancel.";
			close;
		}
		.@i = inarray($@arq_pid, .@pid);
		if (.@i < 0 || .@i >= $@arq_n) {
			mes "Your party is not waiting in the queue.";
			close;
		}
		deletearray $@arq_pid[.@i], 1;
		deletearray $@arq_pref[.@i], 1;
		$@arq_n--;
		setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
		mes "Registration cancelled.";
		close;
	} else if (.@act$ == "status") {
		mes "[Arena Master]";
		mes "^0000FFArenas:^000000";
		for (.@s = 1; .@s <= 4; .@s++) {
			if (!$@ar_state[.@s]) {
				mes "^F77C85~^000000^0152B5Arena " + .@s + "^000000: free";
			} else {
				.@phase$ = ($@ar_state[.@s] == 1) ? "preparing" : (($@ar_state[.@s] == 2) ? "fighting" : "between rounds");
				mes "^F77C85~^000000^0152B5Arena " + .@s + "^000000 [^64CE64" + callfunc("F_GuildTag", $@ar_pa[.@s]) + "^000000 vs ^64CE64" + callfunc("F_GuildTag", $@ar_pb[.@s]) + "^000000] (" + .@phase$ + "): " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + ", round " + $@ar_round[.@s] + ", alive " + $@ar_alive_a[.@s] + "-" + $@ar_alive_b[.@s];
			}
		}
		mes "^0000FFQueue (" + $@arq_n + " waiting):^000000";
		if (!$@arq_n)
			mes "(empty)";
		for (.@i = 0; .@i < $@arq_n; .@i++) {
			.@w$ = ($@arq_pref[.@i] == 0) ? "any free arena" : ("Arena " + $@arq_pref[.@i] + " only");
			mes (.@i + 1) + ". [^FF3A32" + callfunc("F_GuildTag", $@arq_pid[.@i]) + "^000000 waiting] - " + .@w$;
		}
		close;
	} else if (.@act$ == "rules") {
		mes "[Arena Master]";
		mes "- Best of 3 rounds: win 2 to win the match.";
		mes "- A round ends when a whole party is dead, offline or out of the arena.";
		mes "- If the time runs out: more survivors wins, then more HP left.";
		mes "- Everybody is revived and healed between rounds; sides swap.";
		mes "- No repeated job line inside a party (Cardinal = Arch Bishop).";
		close;
	} else if (.@act$ == "watch") {
		mes "[Arena Master]";
		.@m$ = "";
		.@cnt = 0;
		for (.@s = 1; .@s <= 4; .@s++) {

				.@arenaidx[.@cnt] = .@s;
				.@cnt++;
			}
		}
		if (!.@cnt) {
			mes "No match is currently happening.";
			close;
		}
		.@pick = select(.@m$ + ":Cancel");
		if (.@pick > .@cnt) {
			mes "Cancelled.";
			close;
		}
		.@s = .@arenaidx[.@pick - 1];
		$@spec_aid[$@spec_n] = .@myaid;
		$@spec_s[$@spec_n] = .@s;
		$@spec_n++;
		setoption OPTION_INVISIBLE, 1;
		setpcblock PCBLOCK_ATTACK | PCBLOCK_SKILL | PCBLOCK_USEITEM | PCBLOCK_EQUIP | PCBLOCK_CHAT | PCBLOCK_IMMUNE | PCBLOCK_SITSTAND | PCBLOCK_EMOTION, 1;
		#ar_wasspec = 1;
		warp $@map$[.@s], $@specx[.@s], $@specy[.@s];
		close2;
		end;
	} else if (.@act$ == "stopwatch") {
		mes "[Arena Master]";
		callfunc "F_SpecEnd", .@myaid;
		mes "You are no longer watching.";
		close;
	} else if (.@act$ == "gmtools") {
		mes "[Arena Master]";
		if (getgmlevel() < 90) {
			mes "You don't have permission to use this.";
			close;
		}
		switch (select("Force-end a match:Kick a party from queue:Cancel")) {
		case 1:
			.@m$ = "";
			.@cnt = 0;
			for (.@s = 1; .@s <= 4; .@s++) {
				if ($@ar_state[.@s]) {
					.@m$ += ((.@cnt) ? ":" : "") + "Arena " + .@s + " [" + callfunc("F_GuildTag", $@ar_pa[.@s]) + " vs " + callfunc("F_GuildTag", $@ar_pb[.@s]) + "]";
					.@arenaidx[.@cnt] = .@s;
					.@cnt++;
				}
			}
			if (!.@cnt) {
				mes "No arena is currently in a match.";
				close;
			}
			.@pick = select(.@m$ + ":Cancel");
			if (.@pick > .@cnt) {
				mes "Cancelled.";
				close;
			}
			.@s = .@arenaidx[.@pick - 1];
			mapannounce $@map$[.@s], "This match was ended by a GM.", 0, 0xFF0000;
			$@ar_gmreq = .@s;
			donpcevent "Arena2026Ctrl::OnGMForceEnd";
			mes "Arena " + .@s + " was reset and both parties sent back to the lobby.";
			close;
		case 2:
			if (!$@arq_n) {
				mes "The queue is empty.";
				close;
			}
			.@m$ = "";
			for (.@i = 0; .@i < $@arq_n; .@i++)
				.@m$ += ((.@i) ? ":" : "") + callfunc("F_GuildTag", $@arq_pid[.@i]);
			.@pick = select(.@m$ + ":Cancel");
			if (.@pick > $@arq_n) {
				mes "Cancelled.";
				close;
			}
			.@i = .@pick - 1;
			callfunc "F_Tell", $@arq_pid[.@i], "Arena: your registration was cancelled by a GM.";
			deletearray $@arq_pid[.@i], 1;
			deletearray $@arq_pref[.@i], 1;
			$@arq_n--;
			setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
			mes "Removed from the queue.";
			close;
		case 3:
			close;
		}
		end;
	} else {		// "leave"
		close;
	}
	end;

L_Busy:
	.@p = getarg(0);
	if (inarray($@arq_pid, .@p) >= 0 && inarray($@arq_pid, .@p) < $@arq_n)
		return 1;
	for (.@s = 1; .@s <= 4; .@s++) {
		if ($@ar_state[.@s] && ($@ar_pa[.@s] == .@p || $@ar_pb[.@s] == .@p))
			return 1;
	}
	return 0;
}

// -----------------------------------------------------------
// The controller: queue, matches, rounds. One tick per second.
// -----------------------------------------------------------
-	script	Arena2026Ctrl	FAKE_NPC,{
OnInit:
	// ---- settings ----
	.slots = 4;				// arenas (maps)
	.team = 7;				// players per party
	.wins = 2;				// rounds to win a match
	.maxrounds = 5;			// a match never lasts more than this many rounds
	.countdown = 25;		// seconds of preparation before every round (players can move inside their area and buff)
	.roundtime = 30;		// seconds per round (TEMP for testing win-by-timeout; put back to 300 afterwards)
	.between = 8;			// seconds after a round before the next set-up
	.endwait = 10;			// seconds after the match before everybody goes back to the lobby
	.tp_allowed = 0;		// 1 = Teleport / Fly Wing allowed inside the arenas
	$@lobby$ = "hero_ent3";
	$@lx = 69;
	$@ly = 136;
	setarray $@map$[1], "2026arena1", "2026arena2", "2026arena3", "2026arena4";
	// start cell: every one of the 7 players of a side stacks on the very same, verified walkable cell (west / east)
	setarray $@wx[1], 50, 34, 41, 203;
	setarray $@wy[1], 96, 94, 95, 206;
	setarray $@ex[1], 145, 165, 160, 296;
	setarray $@ey[1], 96, 94, 95, 206;
	// where a spectator warps in (separate from the fighters' spawn cells above), all 4 moved per in-game testing.
	setarray $@specx[1], 97, 99, 99, 249;
	setarray $@specy[1], 131, 131, 133, 239;

	for (.@i = 1; .@i <= .slots; .@i++) {
		.@m$ = $@map$[.@i];
		setmapflag .@m$, MF_GVG;
		setmapflag .@m$, MF_PVP_NOGUILD;		// (also honoured on GvG maps) guild mates / allies of the two parties still fight each other
		setmapflag .@m$, MF_NOMEMO;
		setmapflag .@m$, MF_NORETURN;
		setmapflag .@m$, MF_NOBRANCH;
		setmapflag .@m$, MF_NOPENALTY;
		setmapflag .@m$, MF_NOWARPTO;
		setmapflag .@m$, MF_NOTRADE;
		if (!.tp_allowed)
			setmapflag .@m$, MF_NOTELEPORT;
		setmapflagnosave .@m$, $@lobby$, $@lx, $@ly;		// whoever logs out inside comes back in the lobby
		setmapflag .@m$, MF_LOADEVENT;		// so OnPCLoadMapEvent fires here too (defensive, the actual relog lands in the lobby)
	}
	setmapflag $@lobby$, MF_LOADEVENT;		// needed for OnPCLoadMapEvent below to catch a spectator who logged out mid-watch
	// the block below must run exactly once per server boot: mapflags are safe to redo on every @reloadscript, but
	// $@ar_state / $@arq_n are the live match/queue tracking - resetting them on a reload used for something unrelated
	// would silently wipe every match and every queued party (they'd still be physically stuck in the arena maps).
	if (!$@arena2026_booted) {
		$@arena2026_booted = 1;
		for (.@i = 1; .@i <= .slots; .@i++) {
			$@ar_state[.@i] = 0;
			callsub L_Walls, .@i, 1;
		}
		$@arq_n = 0;
		$@spec_n = 0;
		setnpcdisplay "Arena Master", "Arena Master (0 waiting)";
	}
	initnpctimer;
	end;

OnTimer1000:
	if ($@arq_n >= 1)
		callsub L_QueueClean;
	if ($@arq_n >= 2)
		callsub L_Queue;
	for (.@s = 1; .@s <= .slots; .@s++) {
		if ($@ar_state[.@s])
			callsub L_Slot, .@s;
	}
	for (.@i = $@spec_n - 1; .@i >= 0; .@i--) {	// drop spectator entries left behind by a disconnect
		if (!isloggedin($@spec_aid[.@i])) {
			deletearray $@spec_aid[.@i], 1;
			deletearray $@spec_s[.@i], 1;

// Fires on every map load for every player (login, char-select back into the world, warps, everything). A spectator's
// invisibility/blocks are runtime-only and do not survive a fresh session, but OPTION_INVISIBLE itself is one of the
// flags rAthena persists on the character (like riding a Peco) - so without this, going char-select and back would
// bring someone back permanently invisible (blocks gone, invisibility stuck) instead of properly leaving spectator mode.
OnPCLoadMapEvent:
	.@aid = getcharid(3);
	.@i = inarray($@spec_aid, .@aid);
	if (.@i < 0) {
		// #ar_wasspec (account variable) is only ever set by us, right below - never touches a GM's real @hide,
		// which also uses OPTION_INVISIBLE but would never have this flag set.
		if (#ar_wasspec) {
			setoption OPTION_INVISIBLE, 0;
			#ar_wasspec = 0;
		}
		end;
	}
	.@s = $@spec_s[.@i];
	if (strcharinfo(3) != $@map$[.@s] || !$@ar_state[.@s]) {
		deletearray $@spec_aid[.@i], 1;
		deletearray $@spec_s[.@i], 1;
		$@spec_n--;
		setpcblock PCBLOCK_ALL, 0;
		setoption OPTION_INVISIBLE, 0;
		#ar_wasspec = 0;
	} else {
		// still legitimately spectating this same, still-live match - a fresh session lost the block flags, redo them
		setoption OPTION_INVISIBLE, 1;
		setpcblock PCBLOCK_ATTACK | PCBLOCK_SKILL | PCBLOCK_USEITEM | PCBLOCK_EQUIP | PCBLOCK_CHAT | PCBLOCK_IMMUNE | PCBLOCK_SITSTAND | PCBLOCK_EMOTION, 1;
		#ar_wasspec = 1;
	}
	end;

// ---------------------------------------------------------
// queue upkeep: drop any queued party that stopped being match-ready (leader/member disconnected, left the party, died,
// walked out of the lobby, ...) instead of only noticing when it is about to be matched. Runs every tick.
// ---------------------------------------------------------
L_QueueClean:
	.@i = 0;
	while (.@i < $@arq_n) {
		.@e$ = callfunc("F_ArenaCheck", $@arq_pid[.@i], .team, $@lobby$);
		if (.@e$ != "") {
			callfunc "F_Tell", $@arq_pid[.@i], "Arena: your registration was cancelled - " + .@e$;
			deletearray $@arq_pid[.@i], 1;
			deletearray $@arq_pref[.@i], 1;
			$@arq_n--;
			setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
		} else {
			.@i++;
		}
	}
	return;

// ---------------------------------------------------------
// queue -> match
// ---------------------------------------------------------
L_Queue:
	// matches a free arena that currently has 2+ compatible queued parties (preference 0 = any arena, or that arena
	// number) against the first two of them. When several free arenas could host a match right now (e.g. an "any free
	// arena" party could go against a party waiting on arena 3 OR one waiting on arena 4), which one it plays picks a
	// fair, uniformly random arena among all of those candidates - not always the lowest-numbered one.
	// Restarts the scan whenever the queue changes, since a removal or a match shifts every index after it.
	.@changed = 1;
	while (.@changed) {
		.@changed = 0;
		.@nc = 0;
		for (.@s = 1; .@s <= .slots; .@s++) {
			if ($@ar_state[.@s])
				continue;
			.@found = 0;
			for (.@i = 0; .@i < $@arq_n; .@i++) {
				if ($@arq_pref[.@i] == 0 || $@arq_pref[.@i] == .@s)
					.@found++;
			}
			if (.@found >= 2) {
				.@cand[.@nc] = .@s;
				.@nc++;
			}
		}
		if (.@nc) {
			// this build's rand() errors out (silently aborting the whole script) when min==max, so it can not be
			// called as rand(0, 0) - only roll when there is an actual choice to make between 2+ candidate arenas.
			if (.@nc > 1)
				.@s = .@cand[rand(0, .@nc - 1)];
			else
				.@s = .@cand[0];
			.@i1 = -1;
			.@i2 = -1;
			for (.@i = 0; .@i < $@arq_n; .@i++) {
				if ($@arq_pref[.@i] != 0 && $@arq_pref[.@i] != .@s)
					continue;
				if (.@i1 < 0)
					.@i1 = .@i;
				else {
					.@i2 = .@i;
					break;
				}
			}
			.@p1 = $@arq_pid[.@i1];
			.@p2 = $@arq_pid[.@i2];
			.@e1$ = callfunc("F_ArenaCheck", .@p1, .team, $@lobby$);
			.@e2$ = callfunc("F_ArenaCheck", .@p2, .team, $@lobby$);
			if (.@e1$ != "" || .@e2$ != "") {
				if (.@e2$ != "") {		// remove the higher index first so .@i1 stays valid
					callfunc "F_Tell", .@p2, "Arena: your registration was cancelled - " + .@e2$;
					deletearray $@arq_pid[.@i2], 1;
					deletearray $@arq_pref[.@i2], 1;
					$@arq_n--;
				}
				if (.@e1$ != "") {
					callfunc "F_Tell", .@p1, "Arena: your registration was cancelled - " + .@e1$;
					deletearray $@arq_pid[.@i1], 1;
					deletearray $@arq_pref[.@i1], 1;
					$@arq_n--;
				}
				setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
				.@changed = 1;
				continue;
			}
			deletearray $@arq_pid[.@i2], 1;
			deletearray $@arq_pref[.@i2], 1;
			deletearray $@arq_pid[.@i1], 1;
			deletearray $@arq_pref[.@i1], 1;
			$@arq_n -= 2;
			setnpcdisplay "Arena Master", "Arena Master (" + $@arq_n + " waiting)";
			callsub L_Begin, .@s, .@p1, .@p2;
			.@changed = 1;
		}
	}
	return;

	.@s = getarg(0);
	$@ar_pa[.@s] = getarg(1);
	$@ar_pb[.@s] = getarg(2);
	getpartymember getarg(1), 2;
	for (.@k = 0; .@k < .team; .@k++)
		$@ar_ma[.@s * 10 + .@k] = $@partymemberaid[.@k];
	getpartymember getarg(2), 2;
	for (.@k = 0; .@k < .team; .@k++)
		$@ar_mb[.@s * 10 + .@k] = $@partymemberaid[.@k];
	$@ar_round[.@s] = 1;
	$@ar_sc_a[.@s] = 0;
	$@ar_sc_b[.@s] = 0;
	$@ar_fin[.@s] = 0;
	announce "[Arena " + .@s + "] " + callfunc("F_GuildTag", getarg(1)) + " vs " + callfunc("F_GuildTag", getarg(2)) + " - best of 3!", bc_all, 0x00FFAA;
	callsub L_Prep, .@s;
	return;

		break;
	}
	return;

// ---------------------------------------------------------
// members: team 0 = party A, 1 = party B; returns the account id (0 = none)
// ---------------------------------------------------------
L_Member:
	if (getarg(1) == 0)
		return $@ar_ma[getarg(0) * 10 + getarg(2)];
	return $@ar_mb[getarg(0) * 10 + getarg(2)];

// ---------------------------------------------------------
// set-up of a round: revive, heal, clean, warp to the start (sides swap every round), freeze, countdown
// ---------------------------------------------------------
L_Prep:
	.@s = getarg(0);
	callsub L_Walls, .@s, 0;
	.@flip = ($@ar_round[.@s] % 2 == 0) ? 1 : 0;
	.@on0 = 0;
	.@on1 = 0;
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (!.@a || !isloggedin(.@a))
				continue;
			if (!attachrid(.@a))
				continue;
			if (.@t == 0) .@on0++; else .@on1++;
			recovery 0, getcharid(0);
			sc_end SC_ALL;
			percentheal 100, 100;
			if ((.@t ^ .@flip) == 0)
				warp $@map$[.@s], $@wx[.@s], $@wy[.@s];
			else
				warp $@map$[.@s], $@ex[.@s], $@ey[.@s];
			// AP after the warp (a map change is the last thing that could touch it); healap goes through status_heal like every other heal
			healap MaxAp;
			detachrid;
		}
	}
	// a party that is gone loses by forfeit; nobody left ends the match
	if (!.@on0 || !.@on1) {
		callsub L_Forfeit, .@s, .@on0, .@on1;
		return;
	}
	$@ar_alive_a[.@s] = .@on0;
	$@ar_alive_b[.@s] = .@on1;
	$@ar_state[.@s] = 1;
	$@ar_t[.@s] = .countdown;
	mapannounce $@map$[.@s], "Preparation: " + .countdown + " seconds. Move only inside your area and use your buffs.", 0, 0x00FFFF;
	mapannounce $@map$[.@s], "Round " + $@ar_round[.@s] + " - [ " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " ] " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + " [ " + callfunc("F_GuildTag", $@ar_pb[.@s]) + " ]", 0, 0xFFFF00;
	return;

// a party that has nobody left online loses by forfeit; nobody left at all just ends the match (called from the prep
// stage and from the countdown, so a disconnect is never left hanging until the fight starts)
L_Forfeit:
	.@s = getarg(0);
	.@on0 = getarg(1);
	.@on1 = getarg(2);
	$@ar_fin[.@s] = 1;
	if (!.@on0 && .@on1)
		$@ar_sc_b[.@s] = .wins;
	else if (.@on0 && !.@on1)
		$@ar_sc_a[.@s] = .wins;
	callsub L_Walls, .@s, 1;
	callsub L_EndMatch, .@s;
	return;

// members of a team still online (regardless of where they are); used during the countdown, before anyone can die
L_OnlineCount:
	.@s = getarg(0);
	.@c = 0;
	for (.@k = 0; .@k < .team; .@k++) {
		.@a = callsub(L_Member, .@s, getarg(1), .@k);
		if (.@a && isloggedin(.@a))
			.@c++;
	}
	return .@c;

// ---------------------------------------------------------
// one tick of a running arena
// ---------------------------------------------------------
L_Slot:
	.@s = getarg(0);
	.@st = $@ar_state[.@s];
	if (.@st == 1) {
		.@oa = callsub(L_OnlineCount, .@s, 0);
		.@ob = callsub(L_OnlineCount, .@s, 1);
		if (!.@oa || !.@ob) {
			callsub L_Forfeit, .@s, .@oa, .@ob;
		} else if ($@ar_t[.@s] > 0) {
			if ($@ar_t[.@s] <= 5 || $@ar_t[.@s] % 5 == 0)
				mapannounce $@map$[.@s], "The round starts in " + $@ar_t[.@s] + "...", 0, 0xFFFFFF;
			$@ar_t[.@s]--;
		} else {
			callsub L_StartRound, .@s;
		}
	} else if (.@st == 2) {
		$@ar_t[.@s]--;
		.@aa = callsub(L_Alive, .@s, 0);
		.@ab = callsub(L_Alive, .@s, 1);
		if (.@aa != $@ar_alive_a[.@s] || .@ab != $@ar_alive_b[.@s]) {		// live scoreboard: only announces when someone dies / reconnects
			$@ar_alive_a[.@s] = .@aa;
			$@ar_alive_b[.@s] = .@ab;
			mapannounce $@map$[.@s], "[ " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " ] " + .@aa + " - " + .@ab + " [ " + callfunc("F_GuildTag", $@ar_pb[.@s]) + " ] alive", 0, 0x00FFFF;
		}
		if (!.@aa || !.@ab || $@ar_t[.@s] <= 0)
			callsub L_EndRound, .@s, .@aa, .@ab;
	} else if (.@st == 3) {
		if ($@ar_t[.@s] > 0) {
			$@ar_t[.@s]--;
		} else if ($@ar_fin[.@s]) {
			callsub L_Finish, .@s;
		} else {
			callsub L_Prep, .@s;
		}
	}
	return;

L_StartRound:
	.@s = getarg(0);
	callsub L_Walls, .@s, 1;
	$@ar_state[.@s] = 2;
	$@ar_t[.@s] = .roundtime;
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (.@a && isloggedin(.@a) && attachrid(.@a)) {
				showdigit .roundtime, 2;		// live MM:SS clock at the top of the screen, like a PVP room; the client ticks it down on its own
				detachrid;
			}
		}
	}
	mapannounce $@map$[.@s], "FIGHT!", 0, 0xFF0000;
	return;

// members of a team that are alive and still inside the arena map
L_Alive:
	.@s = getarg(0);
	.@c = 0;
	for (.@k = 0; .@k < .team; .@k++) {
		.@a = callsub(L_Member, .@s, getarg(1), .@k);
		if (!.@a || !isloggedin(.@a) || isdead(.@a))
			continue;
		if (attachrid(.@a)) {
			if (strcharinfo(3) == $@map$[.@s])
				.@c++;
			detachrid;
		}
	}
	return .@c;

// sum of HP% of the living members of a team (tie-break when the time runs out)
L_HpSum:
	.@s = getarg(0);
	.@sum = 0;
	for (.@k = 0; .@k < .team; .@k++) {
		.@a = callsub(L_Member, .@s, getarg(1), .@k);
		if (!.@a || !isloggedin(.@a) || isdead(.@a))
			continue;
		if (attachrid(.@a)) {
			if (strcharinfo(3) == $@map$[.@s] && MaxHp > 0)
				.@sum += Hp * 100 / MaxHp;
			detachrid;
		}
	}
	return .@sum;

L_EndRound:
	.@s = getarg(0);
	.@aa = getarg(1);
	.@ab = getarg(2);
	for (.@t = 0; .@t <= 1; .@t++) {		// stop the round clock right as the round ends (it would otherwise keep counting past 0)
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (.@a && isloggedin(.@a) && attachrid(.@a)) {
				showdigit 0;
				detachrid;
			}
		}
	}
	.@win = 0;
	if (.@aa > .@ab)
		.@win = 1;
	else if (.@ab > .@aa)
		.@win = 2;
	else {
		.@ha = callsub(L_HpSum, .@s, 0);
		.@hb = callsub(L_HpSum, .@s, 1);
		if (.@ha > .@hb) .@win = 1;
		else if (.@hb > .@ha) .@win = 2;
	}
	if (.@win == 1) {
		$@ar_sc_a[.@s]++;
		mapannounce $@map$[.@s], "Round " + $@ar_round[.@s] + ": " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " wins the round! [ " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " ] " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + " [ " + callfunc("F_GuildTag", $@ar_pb[.@s]) + " ]", 0, 0x00FF00;
	} else if (.@win == 2) {
		$@ar_sc_b[.@s]++;
		mapannounce $@map$[.@s], "Round " + $@ar_round[.@s] + ": " + callfunc("F_GuildTag", $@ar_pb[.@s]) + " wins the round! [ " + callfunc("F_GuildTag", $@ar_pa[.@s]) + " ] " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + " [ " + callfunc("F_GuildTag", $@ar_pb[.@s]) + " ]", 0, 0x00FF00;
	} else {
		mapannounce $@map$[.@s], "Round " + $@ar_round[.@s] + ": draw - the round is replayed.", 0, 0x00FF00;
	}
	// everybody stays where they are, but nobody can be hurt any more
	if ($@ar_sc_a[.@s] >= .wins || $@ar_sc_b[.@s] >= .wins || $@ar_round[.@s] >= .maxrounds) {
		callsub L_EndMatch, .@s;
	} else {
		$@ar_round[.@s]++;
		$@ar_state[.@s] = 3;
		$@ar_t[.@s] = .between;
		$@ar_fin[.@s] = 0;
	}
	return;

L_EndMatch:
	.@s = getarg(0);
	callsub L_Walls, .@s, 1;
	.@ga$ = callfunc("F_GuildTag", $@ar_pa[.@s]);
	.@gb$ = callfunc("F_GuildTag", $@ar_pb[.@s]);
	.@score$ = "[ " + .@ga$ + " ] " + $@ar_sc_a[.@s] + " - " + $@ar_sc_b[.@s] + " [ " + .@gb$ + " ]";
	if ($@ar_sc_a[.@s] > $@ar_sc_b[.@s]) {
		.@msg$ = .@ga$ + " wins the match! " + .@score$;
		callfunc "F_StatBump", .@ga$, "wins";
		callfunc "F_StatBump", .@gb$, "losses";
	} else if ($@ar_sc_b[.@s] > $@ar_sc_a[.@s]) {
		.@msg$ = .@gb$ + " wins the match! " + .@score$;
		callfunc "F_StatBump", .@gb$, "wins";
		callfunc "F_StatBump", .@ga$, "losses";
	} else {
		.@msg$ = "The match ends in a draw. " + .@score$;
		callfunc "F_StatBump", .@ga$, "draws";
		callfunc "F_StatBump", .@gb$, "draws";
	}
	announce "[Arena " + .@s + "] " + .@msg$, bc_all, 0x00FFAA;
	mapannounce $@map$[.@s], .@msg$, 0, 0x00FF00;
	$@ar_fin[.@s] = 1;
	$@ar_state[.@s] = 3;
	$@ar_t[.@s] = .endwait;
	return;

// after the match: everybody back to the lobby, healed, and the arena is free again
L_Finish:
	.@s = getarg(0);
	for (.@t = 0; .@t <= 1; .@t++) {
		for (.@k = 0; .@k < .team; .@k++) {
			.@a = callsub(L_Member, .@s, .@t, .@k);
			if (!.@a || !isloggedin(.@a) || !attachrid(.@a))
				continue;
			pcblockmove getcharid(3), 0;
			if (strcharinfo(3) == $@map$[.@s]) {
				recovery 0, getcharid(0);
				sc_end SC_ALL;
				percentheal 100, 100;
				warp $@lobby$, $@lx, $@ly;
				healap MaxAp;
			}
			detachrid;
		}
	}
	$@ar_state[.@s] = 0;
	$@ar_pa[.@s] = 0;
	$@ar_pb[.@s] = 0;
	// send home anybody who was spectating this arena (walk backwards: F_SpecEnd deletes from $@spec_* as it goes)
	for (.@i = $@spec_n - 1; .@i >= 0; .@i--) {
		if ($@spec_s[.@i] == .@s)
			callfunc "F_SpecEnd", $@spec_aid[.@i];
	}
	callsub L_Queue;
	return;
}
